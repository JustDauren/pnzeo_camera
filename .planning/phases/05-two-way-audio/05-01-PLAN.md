**audio_codec.py** -- Create a new file implementing pure Python G.711 A-law codec (per AUDI-04). No external dependencies (Python 3.13 removed audioop). Two precomputed 256-entry lookup tables:

1. `ALAW_DECODE_TABLE: list[int]` -- 256 entries mapping A-law byte (0x00-0xFF) to signed 16-bit PCM value (-32768..32767). Standard ITU-T G.711 A-law decoding:
   - Toggle even bits (XOR 0x55)
   - Extract sign bit (bit 7), exponent (bits 4-6), mantissa (bits 0-3)
   - If exponent == 0: linear = (mantissa * 2 + 1) * 2 (shift by 1, bias 1)
   - If exponent != 0: linear = ((mantissa * 2 + 33) << exponent) (shift by exponent, bias 33)
   - Apply sign: negate if sign bit is 0

2. `ALAW_ENCODE_TABLE: list[int]` -- 65536 entries mapping unsigned index (PCM + 32768) to A-law byte. Build by inverting the decode table for the 256 decode values, then fill gaps by finding nearest decoded value for each of the 65536 possible PCM values. Alternatively, implement the standard compression algorithm directly:
   - Take absolute value of sample, clamp to 32767
   - Find exponent (position of highest bit above bit 4, 0-7)
   - Extract 4 mantissa bits below the exponent
   - Combine: sign(1) | exponent(3) | mantissa(4)
   - XOR with 0x55

3. Functions:
   ```python
   def alaw_decode(data: bytes) -> bytes:
       """Decode A-law bytes to 16-bit signed LE PCM. Returns bytes (2x input length)."""
   
   def alaw_encode(pcm_data: bytes) -> bytes:
       """Encode 16-bit signed LE PCM to A-law bytes. Returns bytes (half input length)."""

   def detect_audio_format(drw_payload: bytes) -> dict:
       """Detect audio format from DRW audio packet header.
       
       Returns dict with keys: codec (str), sample_rate (int), channels (int), bits_per_sample (int).
       Parses first 4 bytes of audio DRW payload for format info.
       Camera audio DRW inner header (inferred from APK RTNativeCaller):
         byte 0: codec type (0=PCM, 1=A-law, 2=u-law, 3=ADPCM)
         byte 1: sample_rate indicator (0=8000, 1=16000, 2=32000)
         byte 2: channels (1=mono, 2=stereo)
         byte 3: bits per sample (8 or 16)
       If header is not recognizable, default to: A-law, 8000Hz, mono, 8-bit.
       """
   ```

Use table lookups in both encode and decode, no per-sample arithmetic at runtime. Import struct for PCM byte conversion.

    """Route DRW data by channel."""
    if channel == CH_CMD:
        # CGI command response -- signal _send_cgi
        self._cmd_data = data
        self._cmd_response.set()
    elif channel == CH_AUDIO:
        # Audio stream data -- queue for consumer

            if self._audio_format is None and len(data) > 7:
                # First audio packet -- detect format from inner header
                from .audio_codec import detect_audio_format

                self._audio_queue.put_nowait(data)

                # Drop oldest, keep newest (backpressure)

                    self._audio_queue.get_nowait()

```python
async def start_audio_stream(self) -> bool:
    """Start receiving audio on CH_AUDIO. Returns True if command sent."""
    if not self.connected or not self._transport:
        return False
    # Clear queue and state
    while not self._audio_queue.empty():
        try:
            self._audio_queue.get_nowait()
        except asyncio.QueueEmpty:
            break
    self._audio_format = None
    self._audio_streaming = True
    # Send binary start audio command on CH_CMD
    from .pppp_packets import encode_command, build_drw
    from .const import MSG_START_AUDIO
    cmd = encode_command(MSG_START_AUDIO, b'\x01')  # mode=1 (A-law)
    pkt = build_drw(CH_CMD, cmd, index=self._cmd_seq)
    self._cmd_seq = (self._cmd_seq + 1) % 256
    self._transport.sendto(pkt, (self.host, self._cam_port))
    _LOGGER.info("Audio stream started on %s", self.host)
    return True

async def stop_audio_stream(self) -> None:
    """Stop receiving audio on CH_AUDIO."""
    self._audio_streaming = False
    if self.connected and self._transport:
        from .pppp_packets import encode_command, build_drw
        from .const import MSG_STOP_AUDIO
        cmd = encode_command(MSG_STOP_AUDIO)
        pkt = build_drw(CH_CMD, cmd, index=self._cmd_seq)
        self._cmd_seq = (self._cmd_seq + 1) % 256
        self._transport.sendto(pkt, (self.host, self._cam_port))
    # Drain queue
    while not self._audio_queue.empty():
        try:
            self._audio_queue.get_nowait()
        except asyncio.QueueEmpty:
            break
    self._audio_format = None
    _LOGGER.info("Audio stream stopped on %s", self.host)

async def send_talk_data(self, alaw_data: bytes) -> bool:
    """Send A-law encoded audio data to camera on CH_TALK."""
    if not self.connected or not self._transport:
        return False
    from .pppp_packets import build_drw
    from .const import CH_TALK, AUDIO_FRAME_SIZE
    # Send in AUDIO_FRAME_SIZE chunks
    offset = 0
    while offset < len(alaw_data):
        chunk = alaw_data[offset:offset + AUDIO_FRAME_SIZE]
        pkt = build_drw(CH_TALK, chunk, index=self._cmd_seq)
        self._cmd_seq = (self._cmd_seq + 1) % 256
        self._transport.sendto(pkt, (self.host, self._cam_port))
        offset += AUDIO_FRAME_SIZE
    return True
```
