```python
    cgi = build_cgi_url(CGI_CAMERA_CONTROL, self.username, self.password,

    return bool((r := await self._send_cgi(cgi)) and r.get("success"))
```

   a) `async def get_ircut_params(self) -> dict[str, Any]`:
      - Try `build_cgi_url(CGI_GET_IRCUT, ...)` first
      - If no response (endpoint may not exist on all cameras), return empty dict
      - Store ircut params in `self._camera_params` under ircut namespace

   b) `async def set_ircut_params(self, **kwargs) -> bool`:
      - Accepts: ircut_mode (0=auto, 1=on, 2=off), ircut_sensitivity, ircut_day_start, ircut_day_end, ircut_night_start
      - Uses camera_control.cgi param=14 for simple on/off
      - Uses set_ircut.cgi for full parameter control if available
      - Falls back to camera_control for basic mode setting

   c) `async def set_power_freq(self, freq: int) -> bool`:
      - Calls `self.camera_control(CGI_PARAM_POWER_FREQ, freq)`
      - freq: 0=50Hz, 1=60Hz

   d) `async def set_device_name(self, name: str) -> bool`:
      - Calls `build_cgi_url(CGI_SET_DEVNAME, ..., devname=name)`
      - Returns success boolean

   e) `async def sync_time(self) -> bool`:
      - Gets current time from system: `import datetime; now = datetime.datetime.now()`
      - Builds CGI URL with time params: year, month, day, hour, minute, second, timezone offset
      - Sends via `build_cgi_url(CGI_SET_MOBILETIME or CGI_SET_DATETIME, ..., year=now.year, mon=now.month, day=now.day, hour=now.hour, min=now.minute, sec=now.second)`
      - Returns success boolean

   f) `async def start_recording(self) -> bool`:
      - Sets recording mode to continuous recording
      - Calls `build_cgi_url("set_record_param.cgi", ..., rec_mode=1, rec_channel=1)`
      - Returns success boolean

   ```python
   """Text entities for PNZEO camera."""
   from __future__ import annotations

   from homeassistant.components.text import TextEntity
   from homeassistant.config_entries import ConfigEntry
   from homeassistant.core import HomeAssistant
   from homeassistant.helpers.entity_platform import AddEntitiesCallback

   from .const import DOMAIN
   from .coordinator import PNZEOCoordinator
   from .entity import PNZEOEntity


   async def async_setup_entry(
       hass: HomeAssistant, entry: ConfigEntry, async_add_entities: AddEntitiesCallback
   ) -> None:
       coordinator: PNZEOCoordinator = hass.data[DOMAIN][entry.entry_id]
       async_add_entities([
           PNZEODeviceName(coordinator),
       ])


   class PNZEODeviceName(PNZEOEntity, TextEntity):
       _attr_icon = "mdi:label"
       _attr_native_max = 32

       def __init__(self, coordinator: PNZEOCoordinator) -> None:
           super().__init__(coordinator, "device_name", "Device Name")

       @property
       def native_value(self) -> str | None:
           return self.coordinator.data.get("devname", self.coordinator.device.name)

       async def async_set_value(self, value: str) -> None:
           await self.coordinator.device.client.set_device_name(value)
           await self.coordinator.async_request_refresh()
   ```
